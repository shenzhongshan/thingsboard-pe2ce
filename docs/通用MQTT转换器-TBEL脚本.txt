// 第一步：解析 MQTT 集成上传的 JSON 报文。
// 下面的代码只负责把输入转换成 ThingsBoard 集成要求的统一对象，
// 不会因为温度、电流、电压或计数值的大小而主动丢弃业务数据。

var json = decodeToJson(payload);

// 所有型号均保存设备上报原值，协议键同名写入，不进行重命名、倍率、取整或精度转换。

if (json == null) {
    // 报文不是合法 JSON 时，无法取得设备名称和测点字段。
    // 此时没有办法构造可入库的 ThingsBoard 输出对象，只能结束当前报文。
    return null;
}

if (json.header == null) {
    // header 保存设备定位所需的信息；缺少 header 时无法确定数据属于谁。
    return null;
}

if (json.data == null) {
    // data保存整包协议测点；缺少时没有可写入的遥测。
    return null;
}

var header = json.header;
var data = json.data;

/*
 * 设备标识规则：
 * - 网关聚合报文：header.client 是网关，header.salveid 是实际 KSD 设备；
 * - 设备直连报文：没有 salveid 时，header.client 本身必须是 KSD 设备编号。
 * 两种报文形态都要求最终得到的 deviceName 以 KSD 开头，不再兼容其它名称格式。
 */
var deviceName = header.salveid;

if (deviceName == null || deviceName == "") {
    deviceName = header.client;
}

if (deviceName == null || deviceName == "") {
    // 没有设备名称就无法定位或创建设备，不能把数据写入一个错误设备。
    return null;
}

/* 与波形告警脚本一致：第一个下划线及其后内容为附加标记，不属于平台设备名称。 */
var suffixIndex = deviceName.indexOf("_");
if (suffixIndex >= 0) {
    deviceName = deviceName.substring(0, suffixIndex);
}

if (deviceName.length < 7 || deviceName.substring(0, 3) != "KSD") {
    // 当前现场协议约定所有设备编号都以 KSD 开头，且 KSD 后至少要有四位型号。
    // 非 KSD 名称无法按统一协议推断型号，直接拒绝，避免误把网关或其它名称当成设备。
    return null;
}

/* 使用截断后的设备名称，KSD之后四位直接作为防雷型号；不使用terminal_type。 */
var deviceType = deviceName.substring(3, 7);

var productCategory = "lightning";

    // 型号虽然有值，但不在项目产品映射表中。
    // 未知型号缺少产品名称、一级分类和功能分类，不能合法入库；
    // 因此这里拒绝整条报文，而不是把未知型号误当成某个已知产品。
    return null;
}

    return null;
}

var telemetry = {};

/*
 * 第三步：统一处理一次性上报全部测点的报文。
 * 所有型号（包括DEP4）共用当前型号的协议字段白名单，不保留旧版单点兼容分支。
 * 未配置字段直接忽略；已配置字段的键、值与类型均原样保存。
 */
foreach (entry : data.entrySet()) {
    var sourceKey = entry.getKey();
    var allowed = protocolKeys[sourceKey];
    var sourceValue = entry.getValue();

    // null不作为有效遥测保存；0、负数和小数均原样接收，不进行倍率或精度处理。
    if (allowed == true && sourceValue != null) {
        telemetry[sourceKey] = sourceValue;
    }
}
if (telemetry.size() == 0) {
    // 如果报文只包含协议控制字段、null 字段，或没有任何业务测点，
    // 就没有可写入的遥测内容。此处不是数值过滤，而是避免提交空遥测对象。
    return null;
}

/*
 * attributes 按 TBEL 集成解码器约定写入设备服务器属性。
 * 这里只写入系统要求的三项属性。
 */
var attributes = {
    "productCategory": productCategory,
    "functionCategory": functionCategory,
    "productName": productName
};

return {
    "deviceName": deviceName,
    "deviceType": deviceType,
    "telemetry": telemetry,
    "attributes": attributes
};
